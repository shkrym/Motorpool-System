<script setup>
import { ref, computed } from 'vue'
import { useRouter } from 'vue-router'  
import { supabase } from '../lib/supabase.js'

const router = useRouter() 
const isLogin = ref(true)
const email = ref('')
const password = ref('')
const showPassword = ref(false)
const employeeId = ref('')
const fullName = ref('')
const position = ref('')
const department = ref('')
const phone = ref('')
const officePhone = ref('')
const message = ref('')
const messageType = ref('error')

// password rule: at least 1 uppercase + 1 special char
const passwordRegex = /^(?=.*[A-Z])(?=.*[!*\?#%\-_]).{8,}$/

async function handleSubmit() {
  message.value = ''
  messageType.value = 'error'

  if (!isLogin.value) {
    if (!passwordRegex.test(password.value)) {
      message.value = 'Password must have 8+ chars, 1 uppercase, and 1 special symbol (!*?#%-_)'
      return
    }
  }

  if (isLogin.value) {
    const { error } = await supabase.auth.signInWithPassword({
      email: email.value,
      password: password.value
    })
    if (error) {
      message.value = error.message
      messageType.value = 'error'
    } else {
      messageType.value = 'success'
      message.value = 'Login successful! Redirecting...'
      // ✅ Redirect to dashboard and carry info that login succeeded
      setTimeout(() => {
        router.push({ name: 'Dashboard', query: { login: 'success' } })
      }, 1000)
    }
  } else {
    const { error } = await supabase.auth.signUp({
      email: email.value,
      password: password.value,
      options: {
        data: {
          employee_id: employeeId.value,
          full_name: fullName.value,
          position: position.value,
          department: department.value,
          phone: phone.value,
          office_phone: officePhone.value,
          role: position.value || 'staff'
        }
      }
    })
    
    if (error) {
      message.value = error.message
      messageType.value = 'error'
    } else {
      message.value = 'Registration successful! Check your email to confirm your account.'
      messageType.value = 'success'
    }
  }
}

// Forgot password: sends reset link to email
async function resetPassword() {
  if (!email.value) {
    message.value = 'Please enter your email address first.'
    messageType.value = 'error'
    return
  }
  
  const { error } = await supabase.auth.resetPasswordForEmail(email.value, {
    redirectTo: 'http://localhost:5173/reset-password' // change later when you add reset page
  })
  
  if (error) {
    message.value = error.message
    messageType.value = 'error'
  } else {
    message.value = 'Password reset email sent successfully!'
    messageType.value = 'success'
  }
}
</script>
