<template>
  <div class="auth-page">
    <div class="auth-box">
      <h1>
        {{ isLogin ? 'Login to Motorpool' : 'Register Staff Account' }}
      </h1>

      <form @submit.prevent="handleSubmit">
        <!-- Registration-only fields -->
        <div v-if="!isLogin">
          <input v-model="employeeId" type="text" placeholder="Employee ID" required />
          <input v-model="fullName" type="text" placeholder="Full Name" required />

          <!-- dropdown for position -->
          <select v-model="position" required>
            <option disabled value="">Select Position</option>
            <option value="staff">Staff</option>
            <option value="admin">Admin</option>
          </select>

          <input v-model="department" type="text" placeholder="Department" required />
          <input v-model="phone" type="text" placeholder="Phone (optional)" />
          <input v-model="officePhone" type="text" placeholder="Office Phone (optional)" />
        </div>

        <input v-model="email" type="email" placeholder="University Email" required />

        <!-- Password Field -->
        <div class="password-field">
          <input
            :type="showPassword ? 'text' : 'password'"
            v-model="password"
            placeholder="Password"
            required
          />
          <button type="button" class="toggle-pass" @click="showPassword = !showPassword">
            {{ showPassword ? 'Hide' : 'Show' }}
          </button>
        </div>

        <button type="submit" class="auth-submit-btn">
          {{ isLogin ? 'Login' : 'Register' }}
        </button>
      </form>

      <!-- forgot password link -->
      <div v-if="isLogin" class="forgot">
        <button @click="resetPassword">Forgot Password?</button>
      </div>

      <p class="toggle">
        <span v-if="isLogin">No account?</span>
        <span v-else>Already registered?</span>
        <button @click="isLogin = !isLogin">
          {{ isLogin ? 'Register' : 'Login' }}
        </button>
      </p>

      <p v-if="message" class="error-msg">{{ message }}</p>
    </div>

    <!-- Success popup -->
    <div v-if="popupMessage" class="popup-msg">
      {{ popupMessage }}
    </div>
  </div>
</template>

<script setup>
import { ref } from 'vue'
import { useRouter } from 'vue-router'
import { supabase } from '../lib/supabase.js'

const router = useRouter()

const isLogin = ref(true)
const email = ref('')
const password = ref('')
const showPassword = ref(false)
const employeeId = ref('')
const fullName = ref('')
const position = ref('')
const department = ref('')
const phone = ref('')
const officePhone = ref('')
const message = ref('')
const popupMessage = ref('')

// password rule: at least 1 uppercase + 1 special char
const passwordRegex = /^(?=.*[A-Z])(?=.*[!*\?#%\-_]).{8,}$/

function clearForm() {
  email.value = ''
  password.value = ''
  employeeId.value = ''
  fullName.value = ''
  position.value = ''
  department.value = ''
  phone.value = ''
  officePhone.value = ''
}

async function handleSubmit() {
  message.value = ''

  if (!isLogin.value) {
    if (!passwordRegex.test(password.value)) {
      message.value =
        'Password must have 8+ chars, 1 uppercase, and 1 special symbol (!*?#%-_)'
      return
    }
  }

  if (isLogin.value) {
    const { error } = await supabase.auth.signInWithPassword({
      email: email.value,
      password: password.value
    })
    if (error) {
      message.value = error.message
    } else {
      clearForm()
      showPopup('Login successful!')
      router.push({ name: 'Dashboard' })
    }
  } else {
    const { error } = await supabase.auth.signUp({
      email: email.value,
      password: password.value,
      options: {
        data: {
          employee_id: employeeId.value,
          full_name: fullName.value,
          position: position.value,
          department: department.value,
          phone: phone.value,
          office_phone: officePhone.value,
          role: position.value || 'staff'
        }
      }
    })
    if (error) {
      message.value = error.message
    } else {
      clearForm()
      showPopup('Registration successful! Check your email.')
      isLogin.value = true
    }
  }
}

// Forgot password
async function resetPassword() {
  if (!email.value) {
    message.value = 'Enter your email first.'
    return
  }
  const { error } = await supabase.auth.resetPasswordForEmail(email.value, {
    redirectTo: 'http://localhost:5173/reset-password'
  })
  message.value = error ? error.message : 'Password reset email sent.'
}

// Popup handler
function showPopup(text) {
  popupMessage.value = text
  setTimeout(() => {
    popupMessage.value = ''
  }, 1500)
}
</script>
